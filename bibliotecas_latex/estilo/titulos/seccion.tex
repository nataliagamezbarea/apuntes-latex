\newcommand{\secciongrandeindice}{section}
\newcommand{\secciongrandeestilo}{\Large}
\newcommand{\secciongrandelongitud}{1.1cm}
\newcommand{\secciongrandedisplay}[1]{%
  \sectionheader[\colortitulo]{\secciongrandelongitud}{#1}{\secciongrandeestilo}}

\newcommand{\seccion}[2][]{
  % una subseccion NO cambia de color: se queda con el de su seccion
  \if\relax\detokenize{#1}\relax\else\setcolorapartado{#1}\fi
  \Needspace{5\baselineskip}%
  \vspace{4mm}
  \sectionheader[\colortitulo]{0.4cm}{#2}{\large}\vspace{2mm}}

\newcommand{\secciongrande}[2][]{
  \if\relax\detokenize{#1}\relax\sigseccion\else\setcolorapartado{#1}\fi
  \Needspace{12\baselineskip}%
  \vspace{7mm}
  \ifdefined\phantomsection\phantomsection\fi
  \secciongrandedisplay{#2}%
  \addcontentsline{toc}{\secciongrandeindice}{#2}%
  \vspace{2.5mm}}
