% =====================================================================
%  SUBTITULO — el titulo de dentro de una seccion.
%
%  A diferencia de \seccion y \secciongrande, aqui NO hay barras: es
%  solo un subtitulo, dentro de la seccion.  El color es FIJO (no
%  cambia de seccion en seccion), para que se lea como parte de la
%  seccion que lo contiene.
%
%  \subtitulo{Titulo}
%  \subtitulo[otrocolor]{Titulo}
% =====================================================================
\newcommand{\subtitulo}[2][]{%
  \if\relax\detokenize{#1}\relax%
    \colorlet\subtitulocolor}{secondary}%
  \else%
    \colorlet\subtitulocolor}{#1}%
  \fi
  \par\vspace{4mm}%
  {\headingfontxb\large\color{subtitulocolor}#2}\par
  \vspace{1.5mm}%
}
