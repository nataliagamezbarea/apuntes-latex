\newcommand{\sectionheader}[3][primary]{%
  \setlength{\barleftlen}{#2}%
  \settowidth{\bartextlen}{\headingfontxb\Large #3}%
  \setlength{\barrightlen}{\linewidth}%
  \addtolength{\barrightlen}{-\barleftlen}%
  \addtolength{\barrightlen}{-\bartextlen}%
  \addtolength{\barrightlen}{-0.9cm}%
  \noindent\barrule[#1]{\barleftlen}\hspace{4mm}{\headingfontxb\Large\color{#1} #3}\hspace{4mm}\barrule[#1]{\barrightlen}\par
}

% --- A: MARCADOR (mayúsculas gruesas sobre una franja resaltada, estilo img.1)
\newcommand{\tituloA}[2]{%
  \begin{center}
  \begin{tikzpicture}
    \node (t) [text width=15.5cm, align=center] {{\markerfont\fontsize{23}{27}\selectfont\color{ink} #1}};
    \begin{pgfonlayer}{background}
      \fill[tertiary, rotate around={-1.4:($(t.south west)!0.5!(t.north east)$)}]
        ($(t.south west)+(-0.35,-0.15)$) rectangle ($(t.north east)+(0.35,0.18)$);
    \end{pgfonlayer}
  \end{tikzpicture}
  \end{center}
  \vspace{-1mm}
  \begin{center}\textcolor{accent}{\rule{2.6cm}{2.6pt}}\end{center}
  \vspace{2.5mm}
  \begin{center}{\headingfontxb\fontsize{13}{16}\selectfont\color{primary} #2}\end{center}
  \vspace{5mm}
}

% --- B: MINIMALISTA (título sencillo + subrayado fino, estilo img.2)
\newcommand{\tituloB}[2]{%
  {\headingfontxb\fontsize{22}{22}\selectfont\color{ink} #1}\par
  \vspace{1mm}
  \textcolor{primary}{\rule{4.4cm}{2.2pt}}\par
  \vspace{2mm}
  {\headingfont\fontsize{11}{11}\selectfont\color{primary} #2}\par
  \vspace{6mm}
}

% --- C: CURSIVO + BARRA RESALTADA (estilo img.3)
\newcommand{\tituloC}[1]{%
  \begin{center}
  \begin{tikzpicture}
    \node (t) [text width=15.5cm, align=center] {{\brushfont\fontsize{24}{28}\selectfont\color{ink} #1}};
    \begin{pgfonlayer}{background}
      \fill[tertiary] ($(t.south west)+(-0.25,-0.15)$) rectangle ($(t.south east)+(0.25,0.18)$);
    \end{pgfonlayer}
  \end{tikzpicture}
  \end{center}%
  \vspace{4mm}
}
