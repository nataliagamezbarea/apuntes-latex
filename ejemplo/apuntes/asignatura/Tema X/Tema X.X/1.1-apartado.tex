\secciongrande{Apartado 1 -- Localhost y servidor}

\rectangulosdoble
  {localhost}
  {Funciona solo en tu ordenador. Se apaga cuando apagas el portatil.}
  {servidor}
  {Esta siempre encendido. Recibe peticiones de usuarios externos.}

\imagen[0.55\linewidth]{diagrama.png}

\lista{
  \textbf{Hosting compartido}: tu web comparte servidor con otros sitios.|
  \textbf{VPS}: un servidor virtual propio.|
  \textbf{Servidor dedicado}: un equipo fisico solo para tu aplicacion.|
  \textbf{Nube (IaaS/PaaS)}: recursos virtuales bajo demanda.|
  \textbf{Contenedores}: la aplicacion con sus dependencias.
}
