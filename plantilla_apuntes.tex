\sectionheader{0.8cm}{¿Qué es?}

\highlightpar{El \hl{neuromarketing} combina neurociencia y marketing para
entender \emph{por qué} el cliente compra, no solo \emph{qué} compra.}

\infobox[primary]{%
  \textbf{Idea clave:} las decisiones de compra son mayoritariamente
  \emph{inconscientes} y emocionales antes de ser racionales.}

% ---- Dos tarjetas lado a lado ----
\typecardpair
  {Definición}
  {Uso de técnicas de medición cerebral y conductual para estudiar al consumidor.}
  {Objetivo}
  {Predecir el comportamiento de compra y diseñar mejores experiencias.}

\vspace{3mm}

% ---- Lista con viñetas ----
\begin{bulletlist}
  \item \textbf{Rama funcional:} estudia la activación cerebral ante estímulos.
  \item \textbf{Rama conductual:} observa lo que la gente hace (ojos, clics, compras).
\end{bulletlist}

% ---- Lista de tareas / checklist ----
\begin{tarealist}
  \item Leer el artículo base del curso.
  \item Resumir las técnicas de medición (fMRI, EEG, eye tracking).
\end{tarealist}

\vspace{2mm}

% ---- Nota lateral ----
\tipbox{Para el examen}{Diferencia bien entre neuromarketing \emph{funcional}
y \emph{conductual}; caen preguntas tipo test.}
